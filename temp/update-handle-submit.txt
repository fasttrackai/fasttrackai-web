const handleSubmit = async (e: React.FormEvent) => {
  e.preventDefault();
  setIsSubmitting(true);
  
  try {
    // Send the form data to the backend API
    const response = await fetch('/api/schedule-consultation', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        ...formData,
        appointmentDate: dateSelection.selectedDate,
        appointmentTime: dateSelection.selectedTime,
      }),
    });
    
    if (!response.ok) {
      const errorData = await response.json();
      throw new Error(errorData.message || 'Failed to schedule consultation');
    }
    
    setIsSuccess(true);
  } catch (error) {
    console.error('Error scheduling consultation:', error);
    alert('There was an error scheduling your consultation. Please try again.');
  } finally {
    setIsSubmitting(false);
  }
};
